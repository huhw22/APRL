source cluster.env
/usr/bin/mpic++ -std=c++11 -O3 -Wall -Wextra -Wpedantic -fopenmp \
 -DOMPI_SKIP_MPICXX -DMPICH_SKIP_MPICXX -DAPRL_HAS_OPENMP=1 -DAPRL_HAS_MOVING_WINDOW=1 \
 -I source/aprl/src -I "$APRL_DEPENDENCY_PREFIX/include" \
 source/aprl/verification/performance/cases/longitudinal_momentum_origin/first_step_boundary_probe.cpp \
 build/longitudinal-momentum-origin-20261030/libdirect_eb_core.a \
 "$APRL_DEPENDENCY_PREFIX/lib/libhdf5.so" "$APRL_DEPENDENCY_PREFIX/lib/libyaml-cpp.so.0.8.0" \
 /usr/lib/x86_64-linux-gnu/libfftw3.so -Wl,-rpath,"$APRL_DEPENDENCY_PREFIX/lib" \
 -o build/longitudinal-momentum-origin-20261030/first_step_boundary_probe
